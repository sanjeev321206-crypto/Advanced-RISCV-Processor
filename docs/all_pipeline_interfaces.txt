

============================================================
IF_ID_REGISTER
============================================================

module if_id_register (
    input wire clk,
    input wire reset,
    input wire stall,
    input wire flush,
    input wire [31:0] pc_in,
    input wire [31:0] pc_plus4_in,
    input wire [31:0] instruction_in,
    output reg [31:0] pc_out,
    output reg [31:0] pc_plus4_out,
    output reg [31:0] instruction_out
);

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            pc_out <= 0;
            pc_plus4_out <= 0;
            instruction_out <= 32'h00000013;
        end
        else if (flush) begin
            pc_out <= 0;
            pc_plus4_out <= 0;
            instruction_out <= 32'h00000013;
        end
        else if (!stall) begin
            pc_out <= pc_in;
            pc_plus4_out <= pc_plus4_in;
            instruction_out <= instruction_in;
        end
    end

endmodule

module id_ex_register (
    input wire clk,
    input wire reset,
    input wire flush,

    input wire [31:0] pc_in,
    input wire [31:0] pc_plus4_in,
    input wire [31:0] reg_data1_in,
    input wire [31:0] reg_data2_in,
    input wire [31:0] immediate_in,

    input wire [4:0] rs1_in,
    input wire [4:0] rs2_in,
    input wire [4:0] rd_in,

    input wire [3:0] alu_control_in,

    input wire reg_write_in,
    input wire mem_read_in,
    input wire mem_write_in,
    input wire mem_to_reg_in,
    input wire alu_src_in,
    input wire branch_in,
    input wire jump_in,
    input wire jalr_in,
    input wire lui_in,
    input wire auipc_in,

    output reg [31:0] pc_out,
    output reg [31:0] pc_plus4_out,
    output reg [31:0] reg_data1_out,
    output reg [31:0] reg_data2_out,
    output reg [31:0] immediate_out,

    output reg [4:0] rs1_out,
    output reg [4:0] rs2_out,
    output reg [4:0] rd_out,

    output reg [3:0] alu_control_out,

    output reg reg_write_out,
    output reg mem_read_out,
    output reg mem_write_out,
    output reg mem_to_reg_out,
    output reg alu_src_out,
    output reg branch_out,
    output reg jump_out,
    output reg jalr_out,
    output reg lui_out,
    output reg auipc_out
);

    always @(posedge clk or posedge reset) begin

        if (reset || flush) begin

            pc_out <= 0;
            pc_plus4_out <= 0;
            reg_data1_out <= 0;
            reg_data2_out <= 0;
            immediate_out <= 0;

            rs1_out <= 0;
            rs2_out <= 0;
            rd_out <= 0;

            alu_control_out <= 0;

            reg_write_out <= 0;
            mem_read_out <= 0;
            mem_write_out <= 0;
            mem_to_reg_out <= 0;
            alu_src_out <= 0;
            branch_out <= 0;
            jump_out <= 0;
            jalr_out <= 0;
            lui_out <= 0;
            auipc_out <= 0;

        end

        else begin

            pc_out <= pc_in;
            pc_plus4_out <= pc_plus4_in;
            reg_data1_out <= reg_data1_in;
            reg_data2_out <= reg_data2_in;
            immediate_out <= immediate_in;

            rs1_out <= rs1_in;
            rs2_out <= rs2_in;
            rd_out <= rd_in;

            alu_control_out <= alu_control_in;

            reg_write_out <= reg_write_in;
            mem_read_out <= mem_read_in;
            mem_write_out <= mem_write_in;
            mem_to_reg_out <= mem_to_reg_in;
            alu_src_out <= alu_src_in;
            branch_out <= branch_in;
            jump_out <= jump_in;
            jalr_out <= jalr_in;
            lui_out <= lui_in;
            auipc_out <= auipc_in;

        end

    end

endmodule

module ex_mem_register (
    input wire clk,
    input wire reset,

    input wire [31:0] pc_plus4_in,
    input wire [31:0] alu_result_in,
    input wire [31:0] write_data_in,

    input wire [4:0] rd_in,

    input wire reg_write_in,
    input wire mem_read_in,
    input wire mem_write_in,
    input wire mem_to_reg_in,

    output reg [31:0] pc_plus4_out,
    output reg [31:0] alu_result_out,
    output reg [31:0] write_data_out,

    output reg [4:0] rd_out,

    output reg reg_write_out,
    output reg mem_read_out,
    output reg mem_write_out,
    output reg mem_to_reg_out
);

    always @(posedge clk or posedge reset) begin

        if (reset) begin

            pc_plus4_out <= 0;
            alu_result_out <= 0;
            write_data_out <= 0;

            rd_out <= 0;

            reg_write_out <= 0;
            mem_read_out <= 0;
            mem_write_out <= 0;
            mem_to_reg_out <= 0;

        end

        else begin

            pc_plus4_out <= pc_plus4_in;
            alu_result_out <= alu_result_in;
            write_data_out <= write_data_in;

            rd_out <= rd_in;

            reg_write_out <= reg_write_in;
            mem_read_out <= mem_read_in;
            mem_write_out <= mem_write_in;
            mem_to_reg_out <= mem_to_reg_in;

        end

    end

endmodule

============================================================
FETCH_STAGE
============================================================

module fetch_stage #(
    parameter IMEM_WORDS = 256
)(
    input wire clk,
    input wire reset,
    input wire stall,
    input wire flush,

    input wire redirect,
    input wire [31:0] redirect_target,

    output wire [31:0] pc,
    output wire [31:0] instruction,
    output wire [31:0] pc_plus4
);

    wire [31:0] next_pc_value;

    program_counter pc_unit (
        .clk(clk),
        .reset(reset),
        .enable(!stall),
        .next_pc(next_pc_value),
        .pc(pc)
    );

    instruction_memory #(
        .WORDS(IMEM_WORDS)
    ) instruction_memory_unit (
        .address(pc),
        .instruction(instruction)
    );

    assign pc_plus4 = pc + 32'd4;

    assign next_pc_value =
        redirect ? redirect_target : pc_plus4;

endmodule

DECODE_STAGE
============================================================

module decode_stage (
    input wire clk,
    input wire reset,
    input wire [31:0] pc,
    input wire [31:0] pc_plus4,
    input wire [31:0] instruction,

    input wire wb_reg_write,
    input wire [4:0] wb_rd,
    input wire [31:0] wb_data,

    output wire [31:0] reg_data1,
    output wire [31:0] reg_data2,
    output wire [31:0] immediate,

    output wire [4:0] rs1,
    output wire [4:0] rs2,
    output wire [4:0] rd,
    output wire [2:0] funct3,

    output wire reg_write,
    output wire mem_read,
    output wire mem_write,
    output wire mem_to_reg,
    output wire alu_src,
    output wire branch,
    output wire jump,
    output wire jalr,
    output wire lui,
    output wire auipc,
    output wire [3:0] alu_control
);

    assign rs1 = instruction[19:15];
    assign rs2 = instruction[24:20];
    assign rd = instruction[11:7];
    assign funct3 = instruction[14:12];

    register_file rf (
        .clk(clk),
        .reset(reset),
        .reg_write(wb_reg_write),
        .read_addr1(rs1),
        .read_addr2(rs2),
        .write_addr(wb_rd),
        .write_data(wb_data),
        .read_data1(reg_data1),
        .read_data2(reg_data2)
    );

    immediate_generator imm_gen (
        .instruction(instruction),
        .immediate(immediate)
    );

    control_unit cu (
        .instruction(instruction),
        .reg_write(reg_write),
        .mem_read(mem_read),
        .mem_write(mem_write),
        .mem_to_reg(mem_to_reg),
        .alu_src(alu_src),
        .branch(branch),
        .jump(jump),
        .jalr(jalr),
        .lui(lui),
        .auipc(auipc),
        .alu_control(alu_control)
    );

endmodule


============================================================
EXECUTE_STAGE
============================================================

module execute_stage (
    input wire [31:0] pc,
    input wire [31:0] pc_plus4,
    input wire [31:0] reg_data1,
    input wire [31:0] reg_data2,
    input wire [31:0] immediate,

    input wire [4:0] rs1,
    input wire [4:0] rs2,
    input wire [4:0] rd,

    input wire [3:0] alu_control,
    input wire alu_src,
    input wire branch,
    input wire jump,
    input wire jalr,
    input wire lui,
    input wire auipc,
    input wire [2:0] funct3,

    output wire [31:0] alu_result,
    output wire [31:0] store_data,
    output wire [31:0] branch_target,
    output wire [31:0] jump_target,
    output wire branch_taken,
    output wire [31:0] writeback_data
);

    wire [31:0] alu_operand_b;
    wire [31:0] alu_result_internal;
    wire alu_zero;

    alu_operand_mux operand_mux (
        .reg_data(reg_data2),
        .immediate(immediate),
        .alu_src(alu_src),
        .alu_operand(alu_operand_b)
    );

    alu arithmetic_unit (
        .a(reg_data1),
        .b(alu_operand_b),
        .alu_control(alu_control),
        .result(alu_result_internal),
        .zero(alu_zero)
    );

    branch_comparator comparator (
        .operand_a(reg_data1),
        .operand_b(reg_data2),
        .funct3(funct3),
        .branch_taken(branch_taken)
    );

    assign store_data = reg_data2;

    assign alu_result = lui ? immediate :
                        auipc ? pc + immediate :
                        alu_result_internal;

    assign branch_target = pc + immediate;

    assign jump_target = jalr ?
        ((reg_data1 + immediate) & 32'hFFFFFFFE) :
        (pc + immediate);

    assign writeback_data = jump ? pc_plus4 : alu_result;

endmodule

        .address(address),
